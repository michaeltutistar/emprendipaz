# Comandos para conectar con GitHub
# Reemplaza TU-USUARIO con tu nombre de usuario de GitHub

# 1. Agregar el repositorio remoto
git remote add origin https://github.com/TU-USUARIO/e-learning-platform.git

# 2. Configurar la rama principal
git branch -M main

# 3. Subir el código
git push -u origin main

# 4. Verificar que se subió correctamente
git remote -v
